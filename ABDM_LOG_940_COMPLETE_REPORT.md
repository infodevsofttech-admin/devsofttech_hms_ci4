### 2. Client-Side Display
**File**: `app/Views/partials/abha_verify_modal.php` (Lines 224-237)

**Implementation:**
```javascript
var card = profile.card_base64 || '';
if (card) {
    // Convert to data URL if not already
    var cardSrc = String(card).indexOf('data:') === 0 
        ? card 
        : 'data:' + (profile.card_content_type || 'image/png') + ';base64,' + card;
    
    // Check if it's official ABDM card
    var isOfficial = profile.card_source === 'abdm';
    
    // Display appropriate note
    var note = isOfficial
        ? ''
        : '<div class="small text-warning mt-1">Provisional card generated by the Bridge, not the official ABDM card.</div>';
    
    // Render card image
    $('#abhaVerifyCardWrap').html('<img src="' + cardSrc + '" alt="ABHA card">' + note);
    
    // Enable download link
    $('#abhaVerifyDownloadCard').attr('href', cardSrc).removeClass('d-none');
} else {
    // Fallback if no card data
    $('#abhaVerifyCardWrap').html('<div class="text-center text-muted">Official ABHA card was not returned by the Bridge.</div>');
    $('#abhaVerifyDownloadCard').addClass('d-none');
}
```
